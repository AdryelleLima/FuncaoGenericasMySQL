CREATE DEFINER=`root`@`%` FUNCTION `validar_cpf`(p_cpf VARCHAR(20)) RETURNS varchar(20) CHARSET utf8mb4
    DETERMINISTIC
BEGIN
    DECLARE v_limpo VARCHAR(11);
    DECLARE v_digito1 INT;
    DECLARE v_digito2 INT;
    DECLARE v_soma INT;
    DECLARE v_i INT;
    
    SET v_limpo = REGEXP_REPLACE(p_cpf, '[^0-9]', '');
    
    -- Verifica se tem 11 dígitos ou se é uma sequência repetida inválida (ex: '11111111111')
    IF LENGTH(v_limpo) <> 11 OR v_limpo REGEXP '^(.)\\1+$' THEN
        RETURN 'Invalido';
    END IF;
    
    -- Cálculo do 1º Dígito Verificador
    SET v_soma = 0;
    SET v_i = 1;
    WHILE v_i <= 9 DO
        SET v_soma = v_soma + CAST(SUBSTRING(v_limpo, v_i, 1) AS UNSIGNED) * (11 - v_i);
        SET v_i = v_i + 1;
    END WHILE;
    
    SET v_digito1 = 11 - (v_soma % 11);
    IF v_digito1 >= 10 THEN
        SET v_digito1 = 0;
    END IF;
    
    -- Cálculo do 2º Dígito Verificador
    SET v_soma = 0;
    SET v_i = 1;
    WHILE v_i <= 10 DO
        SET v_soma = v_soma + CAST(SUBSTRING(v_limpo, v_i, 1) AS UNSIGNED) * (12 - v_i);
        SET v_i = v_i + 1;
    END WHILE;
    
    SET v_digito2 = 11 - (v_soma % 11);
    IF v_digito2 >= 10 THEN
        SET v_digito2 = 0;
    END IF;
    
    -- Valida se os dígitos calculados conferem com os informados
    IF SUBSTRING(v_limpo, 10, 1) = CAST(v_digito1 AS CHAR) 
       AND SUBSTRING(v_limpo, 11, 1) = CAST(v_digito2 AS CHAR) THEN
        RETURN 'Válido'; -- Válido
    ELSE
        RETURN 'Inválido'; -- Inválido
    END IF;
END
